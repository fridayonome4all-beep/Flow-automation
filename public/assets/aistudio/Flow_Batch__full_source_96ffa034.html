Flow Batch: full source

Six files. Tap Copy on each one and save it with the exact file name inside one folder called flow-batch.

  1. Create the folder and the six files.
  2. Open your browser’s extensions page and turn on Developer mode.
  3. Choose Load unpacked and pick the folder.
  4. Open Google Flow in a tab, open the extension, add prompts, press Start.

content.js

Runs the queue inside the Flow tab.

(() => {
  if (window.__flowBatch) return;
  window.__flowBatch = true;

  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
  const get = (k) => new Promise((r) => chrome.storage.local.get(k, (v) => r(v[k])));
  const set = (k, v) => new Promise((r) => chrome.storage.local.set({ [k]: v }, r));
  const visible = (el) => {
    const r = el.getBoundingClientRect();
    return r.width > 0 && r.height > 0;
  };
  const DELAYS = { slow: [15, 25], balanced: [6, 10], fast: [1, 3] };
  let busy = false;

  async function log(run, msg) {
    run.log = [...(run.log || []).slice(-40), new Date().toLocaleTimeString() + "  " + msg];
    await set("run", run);
  }

  // ---------- DOM helpers (edit these if Flow's layout changes) ----------
  function clickByText(text) {
    const sel = 'button,[role="button"],[role="option"],[role="menuitem"],[role="tab"],[role="radio"],li,[tabindex]';
    const items = [...document.querySelectorAll(sel)].filter(visible);
    const t = text.toLowerCase();
    const label = (e) => ((e.getAttribute("aria-label") || "") + "|" + e.textContent).toLowerCase();
    const el =
      items.find((e) => e.textContent.trim().toLowerCase() === t) ||
      items.find((e) => (e.getAttribute("aria-label") || "").toLowerCase() === t) ||
      items.find((e) => label(e).includes(t) && e.textContent.length < 60);
    if (el) { el.click(); return true; }
    return false;
  }

  function promptBox() {
    const c = [...document.querySelectorAll('textarea,[contenteditable="true"],[role="textbox"]')].filter(visible);
    c.sort((a, b) => b.getBoundingClientRect().width - a.getBoundingClientRect().width);
    return c[0];
  }

  function setPrompt(box, text) {
    box.focus();
    if (box.tagName === "TEXTAREA" || box.tagName === "INPUT") {
      const proto = box.tagName === "TEXTAREA" ? HTMLTextAreaElement : HTMLInputElement;
      Object.getOwnPropertyDescriptor(proto.prototype, "value").set.call(box, text);
      box.dispatchEvent(new Event("input", { bubbles: true }));
    } else {
      document.execCommand("selectAll", false, null);
      document.execCommand("insertText", false, text);
    }
  }

  function submit(box) {
    const re = /(^|\s)(create|generate|run|send|submit)\b|arrow_forward|arrow_upward/i;
    const br = box.getBoundingClientRect();
    const dist = (e) => {
      const r = e.getBoundingClientRect();
      return Math.hypot(r.x - br.x, r.y - br.y);
    };
    const btns = [...document.querySelectorAll('button,[role="button"]')]
      .filter(visible)
      .filter((b) => !b.disabled && b.getAttribute("aria-disabled") !== "true")
      .filter((b) => re.test((b.getAttribute("aria-label") || "") + " " + b.textContent.trim()))
      .sort((a, b) => dist(a) - dist(b));
    if (btns[0]) { btns[0].click(); return true; }
    box.dispatchEvent(new KeyboardEvent("keydown", { key: "Enter", code: "Enter", keyCode: 13, bubbles: true }));
    return false;
  }

  function media() {
    const seen = new Set();
    return [...document.querySelectorAll("img,video")]
      .filter((e) => {
        const r = e.getBoundingClientRect();
        return r.width > 120 && r.height > 120;
      })
      .map((e) => ({ src: e.currentSrc || e.src, video: e.tagName === "VIDEO" }))
      .filter((m) => m.src && !m.src.startsWith("data:image/svg") && !seen.has(m.src) && seen.add(m.src));
  }

  async function waitForNew(known, count, timeoutMs) {
    const start = Date.now();
    let found = [];
    while (Date.now() - start < timeoutMs) {
      if (await get("stop")) return found;
      found = media().filter((m) => !known.has(m.src));
      if (found.length >= count) { await sleep(3000); return media().filter((m) => !known.has(m.src)).slice(0, count); }
      await sleep(2000);
    }
    return found;
  }

  async function save(item, name) {
    const ext = item.video ? "mp4" : "png";
    if (item.src.startsWith("blob:") || item.src.startsWith("data:")) {
      const blob = await (await fetch(item.src)).blob();
      const a = document.createElement("a");
      a.href = URL.createObjectURL(blob);
      a.download = name + "." + ext;
      document.body.appendChild(a); a.click(); a.remove();
      setTimeout(() => URL.revokeObjectURL(a.href), 10000);
    } else {
      chrome.runtime.sendMessage({ type: "download", url: item.src, filename: name + "." + ext });
    }
  }

  const slug = (s) => s.toLowerCase().replace(/[^a-z0-9]+/g, "-").split("-").filter(Boolean).slice(0, 5).join("-");

  // ---------- queue ----------
  async function applySettings(run) {
    const o = run.cfg[run.cfg.mode];
    const wanted = [o.model, o.aspect, o.duration].filter(Boolean);
    for (const w of wanted) {
      const ok = clickByText(w);
      await log(run, ok ? "Selected " + w : "Could not find '" + w + "' - set it by hand once");
      await sleep(600);
    }
    const n = o.count;
    if (n && !(clickByText("x" + n) || clickByText("\u00d7" + n))) await log(run, "Could not find x" + n + " - set it by hand once");
  }

  async function finish(run, status) {
    run.status = status;
    await log(run, status === "done" ? "Finished" : "Stopped");
    const hist = (await get("hist")) || [];
    hist.unshift({ ts: Date.now(), status, total: run.prompts.length, done: run.done, prompts: run.prompts });
    await set("hist", hist.slice(0, 10));
  }

  async function runQueue() {
    if (busy) return;
    busy = true;
    try {
      let run = await get("run");
      if (!run || run.status !== "running") return;
      if (!run.applied) { await applySettings(run); run.applied = true; await set("run", run); }
      const known = new Set(media().map((m) => m.src));
      const o = run.cfg[run.cfg.mode];
      const count = parseInt(o.count || "1", 10);
      const timeout = run.cfg.mode === "video" ? 12 * 60000 : 3 * 60000;

      while (run.index < run.prompts.length) {
        if (await get("stop")) { await finish(run, "stopped"); return; }
        const text = run.prompts[run.index];
        const num = String(run.index + 1).padStart(3, "0");
        await log(run, "Prompt " + num + ": " + text.slice(0, 40));
        try {
          const box = promptBox();
          if (!box) throw new Error("prompt box not found");
          setPrompt(box, text);
          await sleep(700);
          submit(box);
          const got = await waitForNew(known, count, timeout);
          if (!got.length) throw new Error("no result before timeout");
          for (let i = 0; i < got.length; i++) {
            known.add(got[i].src);
            const name = run.cfg.naming === "slug" ? num + "-" + slug(text) + "_v" + (i + 1) : num + "_v" + (i + 1);
            await save(got[i], name);
          }
          run.done++;
        } catch (e) {
          run.failed = (run.failed || 0) + 1;
          await log(run, "Failed " + num + ": " + e.message);
        }
        run.index++;
        await set("run", run);
        if (run.index < run.prompts.length) {
          const [lo, hi] = run.cfg.speed === "custom"
            ? [run.cfg.customDelay, run.cfg.customDelay]
            : DELAYS[run.cfg.speed] || DELAYS.balanced;
          await sleep((lo + Math.random() * (hi - lo)) * 1000);
        }
      }
      await finish(run, "done");
    } finally {
      busy = false;
    }
  }

  chrome.runtime.onMessage.addListener((msg, _s, reply) => {
    if (msg && msg.type === "start") {
      (async () => {
        await set("stop", false);
        await set("run", {
          status: "running", prompts: msg.prompts, cfg: msg.cfg,
          index: 0, done: 0, failed: 0, applied: false, log: []
        });
        runQueue();
        reply({ ok: true });
      })();
      return true;
    }
    if (msg && msg.type === "ping") reply({ ok: true });
  });

  // resume after page reload
  get("run").then((r) => { if (r && r.status === "running") setTimeout(runQueue, 4000); });
})();

popup.js

Panel logic: settings, prompts, history.

const OPTS = {
  image: { model: ["Nano Banana Pro", "Nano Banana 2.1", "Nano Banana Lite"], aspect: ["16:9", "4:3", "1:1", "3:4", "9:16"], count: ["1", "2", "3", "4"] },
  video: { model: ["Veo 3.1 Fast", "Veo 3.1 Quality", "Veo 3.1 Lite"], aspect: ["16:9", "9:16"], duration: ["4s", "6s", "8s"], count: ["1", "2", "3", "4"] }
};
const TITLES = { model: "Model", aspect: "Aspect ratio", count: "Outputs per prompt", duration: "Duration" };
let cfg = {
  mode: "image", speed: "balanced", customDelay: 10, naming: "num",
  image: { model: "Nano Banana Pro", aspect: "16:9", count: "1" },
  video: { model: "Veo 3.1 Fast", aspect: "16:9", duration: "8s", count: "1" }
};
const $ = (id) => document.getElementById(id);
const store = (o) => chrome.storage.local.set(o);

function chips(el, items, value, pick) {
  el.innerHTML = "";
  items.forEach(([val, label]) => {
    const b = document.createElement("button");
    b.textContent = label; b.className = val === value ? "on" : "";
    b.onclick = () => { pick(val); store({ cfg }); draw(); };
    el.appendChild(b);
  });
}

function draw() {
  document.querySelectorAll(".tabs button").forEach((b) => b.classList.toggle("on", b.dataset.mode === cfg.mode));
  const box = $("opts"); box.innerHTML = "";
  Object.entries(OPTS[cfg.mode]).forEach(([key, list]) => {
    const h = document.createElement("h2"); h.textContent = TITLES[key];
    const c = document.createElement("div"); c.className = "chips";
    chips(c, list.map((v) => [v, key === "count" ? "x" + v : v]), cfg[cfg.mode][key], (v) => (cfg[cfg.mode][key] = v));
    box.append(h, c);
  });
  chips($("speed"), [["slow", "Slow"], ["balanced", "Balanced"], ["fast", "Fast"], ["custom", "Custom"]], cfg.speed, (v) => (cfg.speed = v));
  chips($("naming"), [["num", "001_v1"], ["slug", "001-prompt_v1"]], cfg.naming, (v) => (cfg.naming = v));
  $("customRow").hidden = cfg.speed !== "custom";
  $("customDelay").value = cfg.customDelay;
}

const parse = () => $("prompts").value.split(/\n\s*\n/).map((s) => s.trim()).filter(Boolean);
function updateCount() { $("count").textContent = parse().length + " prompts"; store({ draft: $("prompts").value }); }

function renderRun(run) {
  if (!run) return;
  const total = run.prompts.length, n = run.index;
  $("state").textContent = run.status === "running" ? "Running " + n + "/" + total : run.status === "done" ? "Done" : "Stopped";
  $("state").className = "pill" + (run.status === "running" ? " run" : "");
  $("fill").style.width = (total ? (n / total) * 100 : 0) + "%";
  $("log").textContent = (run.log || []).join("\n");
  $("log").scrollTop = 1e6;
}

function renderHist(hist) {
  const box = $("hist"); box.innerHTML = "";
  (hist || []).forEach((h) => {
    const d = document.createElement("div"); d.className = "h";
    const t = document.createElement("span");
    t.textContent = new Date(h.ts).toLocaleString() + " - " + h.done + "/" + h.total + " " + h.status;
    const b = document.createElement("button"); b.textContent = "Load";
    b.onclick = () => { $("prompts").value = h.prompts.join("\n\n"); updateCount(); };
    d.append(t, b); box.appendChild(d);
  });
}

async function start() {
  const prompts = parse();
  if (!prompts.length) return ($("log").textContent = "Add at least one prompt.");
  const tabs = await chrome.tabs.query({ url: "https://labs.google/*" });
  if (!tabs.length) return ($("log").textContent = "No Google Flow tab found. Open Flow in a tab, then press Start.");
  const tab = tabs.find((t) => t.active) || tabs[0];
  cfg.customDelay = parseInt($("customDelay").value, 10) || 10;
  store({ cfg });
  chrome.tabs.sendMessage(tab.id, { type: "start", prompts, cfg }, (res) => {
    if (chrome.runtime.lastError) $("log").textContent = "Reload the Flow tab once, then press Start.";
  });
}

document.querySelectorAll(".tabs button").forEach((b) => (b.onclick = () => { cfg.mode = b.dataset.mode; store({ cfg }); draw(); }));
$("prompts").oninput = updateCount;
$("clear").onclick = () => { $("prompts").value = ""; updateCount(); };
$("file").onchange = async (e) => { const f = e.target.files[0]; if (f) { $("prompts").value = await f.text(); updateCount(); } };
$("start").onclick = start;
$("stop").onclick = () => store({ stop: true });
$("customDelay").onchange = () => { cfg.customDelay = parseInt($("customDelay").value, 10) || 10; store({ cfg }); };

chrome.storage.onChanged.addListener((c) => { if (c.run) renderRun(c.run.newValue); if (c.hist) renderHist(c.hist.newValue); });
chrome.storage.local.get(["cfg", "draft", "run", "hist"], (s) => {
  if (s.cfg) cfg = { ...cfg, ...s.cfg };
  if (s.draft) $("prompts").value = s.draft;
  draw(); updateCount(); renderRun(s.run); renderHist(s.hist);
});
